# Spawn vehicle, synced (docs/debug-menu.md "Spawn vehicle", lua/coop/debug_vspawn.lua) in mode=play with the lab key
# named (labsetting debug_vspawn=1: off unless named until this run shows it on both copies). Afghanistan free roam,
# the Wakh Sind road (the seats scenarios' flat spot), both players facing east.
#   A  the host presses Spawn jeep: one of the area's own light vehicles comes 6 m in front of the host, facing east, on
#      both copies (host: event=vspawn; step=pick / apply / placed ok=1; guest: step=rx kind=2 / apply / placed ok=1, the
#      same locator); frames on both screens (the same jeep at the same spot).
#   B  the guest presses Spawn truck: its ask (kind 1) reaches the host, which brings a truck 6 m in front of the GUEST
#      (step=pick ... for=guest) and answers kind 3; the guest applies the same locator; frames on both screens.
#   C  the host drives the spawned jeep with the guest riding: the host gets in beside it (goto_vehicle jeep: the
#      nearest jeep, the spawned one), then the guest beside the host's jeep (the door receipt is lab only); the host
#      holds the throttle; frames on both screens while moving.
#   D  a refusal: a press while riding is refused (busy, or the menu's own on-foot guard), with note 20's toast.
# Run (hidden worker, under the verify lock, MGS_GAME_LUA_DIR unset):
#   pwsh -File tools/worker/Invoke-WorkerScenario.ps1 -ScenarioPath tools/worker/scenarios/r1101-vspawn-play.txt
#        -PartnerProfile native -NativeStage gun -BuildProfile research -SaveSeed afgh-ledge -Mode play -WanSim bad
# NEVER: event=vspawn; step=error, step=placed ... ok=0, bad_op, wrong_dir, unknown_op, GameOver.
labsetting debug_vspawn=1
labsetting nametag=1
title 200
waitgameplay host 120
waitgameplay guest 120
waitlog host 'event=paired' 120
waitlog guest 'event=paired' 120
wait 3000
both debug godmode 1
host debug time 43200
host debug weather sunny
host debug clear_enemies 100
host debug teleport 597.0 320.5 1157.5 90
guest debug teleport 597.0 320.5 1169.5 90
wait 5000
both debug godmode 1
stance host stand
stance guest stand
host run vspawn_status
host expect ^vspawn;on=1;role=host;mission=30010;recent=none;moving=0;ask=0$
guest run vspawn_status
guest expect ^vspawn;on=1;role=guest;mission=30010;
both capture a-before
# --- A: the host's jeep ---------------------------------------------------------------------------------------------
logmark both
host run vspawn lv
host expect ^vspawn;class=lv;loc=veh_lv_\d{4};sent=1$
waitlog host 'world_lua=event=vspawn; step=pick; loc=veh_lv_[0-9]{4}; id=[0-9]+; alive=1;' 5
waitlog host 'world_lua=event=vspawn; step=apply; loc=veh_lv_[0-9]{4}; id=[0-9]+; via=setpos; by=self;' 5
waitlog guest 'world_lua=event=vspawn; step=rx; kind=2;' 10
waitlog guest 'world_lua=event=vspawn; step=apply; loc=veh_lv_[0-9]{4}; id=[0-9]+; via=(setpos|respawn); by=host;' 10
waitlog host 'world_lua=event=vspawn; step=placed; loc=veh_lv_[0-9]{4}; id=[0-9]+; d_m=[0-9.]+; dy_m=[-0-9.]+; ok=1; by=self' 10
waitlog guest 'world_lua=event=vspawn; step=placed; loc=veh_lv_[0-9]{4}; id=[0-9]+; d_m=[0-9.]+; dy_m=[-0-9.]+; ok=1; by=host' 10
wait 1500
both capture a-jeep
host run vspawn_status
guest run vspawn_status
# --- B: the guest's truck -------------------------------------------------------------------------------------------
logmark both
guest run vspawn trc
guest expect ^vspawn_asked;class=trc;seq=\d+$
waitlog host 'world_lua=event=vspawn; step=rx; kind=1;' 10
waitlog host 'world_lua=event=vspawn; step=pick; loc=veh_trc_[0-9]{4}; id=[0-9]+; alive=[01]; d=[0-9]+; for=guest' 5
waitlog host 'world_lua=event=vspawn; step=tx; kind=3;' 5
waitlog guest 'world_lua=event=vspawn; step=apply; loc=veh_trc_[0-9]{4}; id=[0-9]+; via=(setpos|respawn); by=host;' 10
waitlog guest 'world_lua=event=vspawn; step=placed; loc=veh_trc_[0-9]{4};.*ok=1; by=host' 10
waitlog host 'world_lua=event=vspawn; step=placed; loc=veh_trc_[0-9]{4};.*ok=1; by=self' 10
wait 1500
both capture b-truck
host run vspawn_status
guest run vspawn_status

logmark both

wait 2500

wait 3500

host expect ^vehicle;id=\d+;seats=\d+;riders=.*localSeat=1;
guest debug goto_vehicle jeep
wait 2500

wait 4500
guest run vehicle_seats
both capture c-seated

host hold w 1500
host release

both burst c-drive 6 400

host release

both capture c-after
