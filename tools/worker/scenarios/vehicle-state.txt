# Expected (world_lua lines read "event=vehicle_state; step=...", ';' expanded to '; '):
#   both   step=on; mission=30020; role=..; named=N; lrrp=7 (the seven LRRP trucks veh_trc_0000..0006)
#   host   step=sweep; then step=sweep_done; sent=N (every free vehicle once after PAIRED); guest step=same; id=25608;
#          from=25608 (the copies agree at the start: nothing moves)
#   push   host vehicle_push 25608 3 0 (the lab lever: this copy only, a stand-in for a shove) -> host step=tx;
#          id=25608; why=moved; d_mm=~3000 -> guest step=applied; id=25608; from=25608; moved=1; d_mm=~3000; err_mm<1000
#          within 2 s; frames m4-pushed (the jeep 3 m on, on both screens); vehicle_state_status pos on both
#   guest push  guest vehicle_push 25608 0 3 (this copy only) -> within the 10 s refresh guest step=applied; ...; f=5
#          (the host's refresh puts it back); frames m4-guest-pushed (diverged) and m4-corrected (agree again)
#   drive  the guest boards and drives the jeep away, stops and gets off: host step=skip; reason=partner_ride;
#          id=25608 while it drives (no VEHICLE_STATE against RIDE_FAST), step=resume ~10 s after the get-off, then
#          its copy (left where RIDE_FAST put it) is sent: guest step=same or applied with a small d_mm; frames m4-left
#   damage the host shoots the parked jeep: host step=tx; id=25608; why=life (or state); guest step=applied; ...;
#          moved=0; ...; life=<host>; was=<guest> (SetBodyLife read back); frames m4-damaged. Then the lever
#          vehicle_life 25608 200 on the host: guest step=applied; ...; life=200; frames m4-lowlife-NN (smoke or fire
#          on both screens is the burning evidence; the state words sv / here_sv name the bits)
#   E5 wreck  host vehicle_life 25608 0 (the engine's own destruction on the host's copy, a stand-in for a real one,
#          run d407f115) -> host vehicle_gone step=watch; m=VehicleBroken; cause=2 -> tx -> guest step=applied; id=25608;
#          via=wreck; src=rx; ok=1 (SetBodyLife 0 there: a wreck, not a Despawn); frames m4-wreck-NN (both wrecks).
#          Then the guest's own destruction of the next jeep (two shots, then its lever within 10 s): guest step=watch;
#          m=VehicleBroken; cause=2 -> req -> host step=applied; via=wreck; src=req; ok=1; frames m4-req-wreck-NN.
# NEVER: step=error on either copy; a guest step=applied while the guest rides the jeep; a host step=tx for 25608
#   between the drive's skip and resume lines; viewer_changed; world=... or partner_... faults.
