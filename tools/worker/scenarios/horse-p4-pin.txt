# Horse probe P4 (the horse plan, section 4, lab only): a rider state on body 1 (the DM overlay probe
# stateRideOnIdle, lab key partner_native_overlay=debug, as partner-mount.txt Part 5), then body 1 warped every frame to
# the partner's point raised 1.2 m (horse_probe pin, under the CARGO_V2 hold so the partner driver stays off body 1).
# The frames decide whether a frozen seated pose survives the per-tick Warp. Both orders. Part 3 of session A.
labsetting partner_native_overlay=debug
logmark both
# --- guest watches host's body 1
stance host stand
stance guest stand
placebehind guest host 7 2
wait 3000
guest mouse 300 0
wait 800
guest capture p4-guest-before
guest run overlay_state stateRideOnIdle 1 6000
guest expect ^overlay_queued;state=stateRideOnIdle$
wait 800
guest burst p4-guest-dm-ground 2 400
guest run horse_probe pin 120 4000
guest expect horse_probe=pin;step=start
wait 300
guest burst p4-guest-pin-raised 4 500
waitlog guest 'world_lua=horse_probe=pin; step=end;' 8
wait 2500
guest capture p4-guest-after
# --- host watches guest's body 1
stance host stand
stance guest stand
placebehind host guest 7 2
wait 3000
host mouse 300 0
wait 800
host capture p4-host-before
host run overlay_state stateRideOnIdle 1 6000
host expect ^overlay_queued;state=stateRideOnIdle$
wait 800
host burst p4-host-dm-ground 2 400
host run horse_probe pin 120 4000
host expect horse_probe=pin;step=start
wait 300
host burst p4-host-pin-raised 4 500
waitlog host 'world_lua=horse_probe=pin; step=end;' 8
wait 2500
host capture p4-host-after
both capture p4-end
