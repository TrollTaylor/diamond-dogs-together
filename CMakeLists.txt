# The build id both players' link compares (docs/two-pc-play.md "Versions"):
# the version plus the 7-character commit, e.g. 0.12.1+f64a8e9 (14 of the 16
# bytes of the HELLO build field). Without git it stays the plain version.
set(MGS_COOP_BUILD_ID "${PROJECT_VERSION}")
find_package(Git QUIET)
if(GIT_FOUND)
  execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse --short=7 HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
                  OUTPUT_VARIABLE _mgs_git_hash OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET RESULT_VARIABLE _mgs_git_result)
  if(_mgs_git_result EQUAL 0 AND _mgs_git_hash MATCHES "^[0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f]$")
    set(MGS_COOP_BUILD_ID "${PROJECT_VERSION}+${_mgs_git_hash}")
  endif()
endif()
message(STATUS "mgs-coop build id ${MGS_COOP_BUILD_ID}")

    file(READ "${_mgs_world_path}" _mgs_world_text)

    endif()

      string(SUBSTRING "${_mgs_world_text}" 0 15000 _mgs_world_head)

        message(FATAL_ERROR "lua/coop/world_${_mgs_world_file}.lua has no line break in its first 15000 bytes")
      endif()
      math(EXPR _mgs_world_split "${_mgs_world_split} + 1")
      string(SUBSTRING "${_mgs_world_text}" 0 ${_mgs_world_split} _mgs_world_a)
      string(SUBSTRING "${_mgs_world_text}" ${_mgs_world_split} -1 _mgs_world_b)
      string(LENGTH "${_mgs_world_b}" _mgs_world_b_length)
      if(_mgs_world_b_length GREATER 16000)
        message(FATAL_ERROR "lua/coop/world_${_mgs_world_file}.lua second literal is ${_mgs_world_b_length} bytes; the limit is 16000")
      endif()
      set(_mgs_world_text "${_mgs_world_a})MGS_WORLD\" R\"MGS_WORLD(${_mgs_world_b}")
    endif()
    set(MGS_WORLD_${_mgs_world_part}_SOURCE "${_mgs_world_text}")
  endforeach()
